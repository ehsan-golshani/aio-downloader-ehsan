            for (let p of patterns) { let m = t.match(p); if (m) for (let c of m) { let d = detectConfig(c); if (d && !configs.some(x => x.content === c)) configs.push({ content: c, ...d }); } }

        function extractMTProxy(t) {
            let lines = t.split('\n'), server = null, port = null, secret = null;
            for (let l of lines) { 
                let sm = l.match(/Server:\s*(\S+)/i); if (sm) server = sm[1];
                let pm = l.match(/Port:\s*(\d+)/i); if (pm) port = pm[1];
                let sem = l.match(/Secret:\s*(\S+)/i); if (sem) secret = sem[1];
            }
            if (server && port) { 
                let sh = secret || ''; 
                if (sh && !sh.startsWith('ee') && sh.length === 32) sh = 'ee' + sh; 
                return { server, port, secret: sh, link: `tg://proxy?server=${server}&port=${port}&secret=${sh}` }; 
            }
            return null;
        }

        function parseMessagesFromMd(md) {
            let msgs = [];
            let blocks = md.split(/\n(?=## [A-Za-z0-9_]+ — post \d+)/);
            
            for (let b of blocks) {
                let h = b.match(/^## ([A-Za-z0-9_]+) — post (\d+)/m);
                if (!h) continue;
                
                let channel = h[1];
                let postId = parseInt(h[2]);
                
                let images = [];
                let imgMatches = [...b.matchAll(/<img src="([^"]+)" alt="Photo">/g)];
                for (let match of imgMatches) {
                    images.push({ type: 'image', url: fixMediaUrl(match[1]) });
                }
                
                let videos = [];
                let videoMatches = [...b.matchAll(/<video src="([^"]+)"[^>]*>/g)];
                for (let match of videoMatches) {
                    videos.push({ type: 'video', url: fixMediaUrl(match[1]) });
                }
                
                let files = [];
                let fileMatches = [...b.matchAll(/class="file-link"[^>]*>📎 ([^<]+)<\/a>/g)];
                for (let match of fileMatches) {
                    files.push({ type: 'file', name: match[1], url: null });
                }
                
                let text = '';
                let div = b.match(/<div dir='rtl'[^>]*>([\s\S]*?)<\/div>/);
                if (div) {
                    text = div[1];
                    text = text.replace(/<br\s*\/?>/g, '\n').replace(/<[^>]+>/g, '').replace(/&nbsp;/g, ' ').replace(/&quot;/g, '"').replace(/&amp;/g, '&').trim();
                } else {
                    text = b.replace(/<[^>]+>/g, '').replace(/## .+\n/, '').trim();
                }
                
                let mtproxy = extractMTProxy(b);
                let configs = extractConfigs(b + ' ' + text);
                
                let date = '';
                let dm = b.match(/📅 (?:آخرین )?بروزرسانی: ([\d\/]+) (\d+:\d+)/);
                if (dm) {
                    date = `${dm[1]} ${dm[2]}`;
                } else {
                    let pageDate = b.match(/# آرشیو کانال .+? - صفحه \d+\n\n📅 آخرین بروزرسانی: ([\d\/]+) (\d+:\d+)/);
                    if (pageDate) date = `${pageDate[1]} ${pageDate[2]}`;
                }
                
                msgs.push({ channel, postId, date, text, images, videos, files, mtproxy, configs });
            }
            
            msgs.sort((a, b) => b.postId - a.postId);
            return msgs;
        }

                isNearBottom = false;

                loadOldMessages();

        function showLog(id) { var b = $(id); if (b) b.style.display = 'block'; }
        function clearLog(id) {
            var b = $(id); if (!b) return;
            var h = b.querySelector('.log-header'); b.innerHTML = ''; if (h) b.appendChild(h);
            b.style.display = 'none';
        }
        function addLog(id, msg, cls) {
            var b = $(id); if (!b) return;
            b.style.display = 'block';
            var t = new Date().toLocaleTimeString('fa-IR');
            var c = cls ? ' log-' + cls : '';
            var safeMsg = msg.replace(/[&<>]/g, function(m) {
                if (m === '&') return '&amp;';
                if (m === '<') return '&lt;';
                if (m === '>') return '&gt;';
                return m;
            });
            b.innerHTML += '<span class="log-time">[' + t + ']</span><span class="' + c + '">' + safeMsg + '</span>\n';
            b.scrollTop = b.scrollHeight;
        }
        window.copyLog = function(id) {
            var b = $(id); if (!b) return;
            var txt = b.textContent || '';
            if (navigator.clipboard) { navigator.clipboard.writeText(txt).then(function() { toast('✅ کپی شد'); }); }
            else { var ta = document.createElement('textarea'); ta.value = txt; ta.style.position = 'fixed'; ta.style.opacity = '0'; document.body.appendChild(ta); ta.select(); document.execCommand('copy'); document.body.removeChild(ta); toast('✅ کپی شد'); }
        };
        
        function mainApi(path, method, body) {
            if (!method) method = 'GET';
            if (!S.token) return Promise.reject(new Error('توکن تنظیم نشده'));
            var headers = { 'Authorization': 'token ' + S.token, 'Accept': 'application/vnd.github.v3+json' };
            if (body) headers['Content-Type'] = 'application/json';
            return fetch('https://api.github.com' + path, { method: method, headers: headers, body: body ? JSON.stringify(body) : null })
                .then(function(res) {
                    if (res.status === 401) throw new Error('توکن نامعتبر');
                    if (res.status === 403) throw new Error('Rate limit');
                    if (!res.ok) throw new Error('HTTP ' + res.status);
                    return res;
                });
        }
        
        function saveSettings() {
            var url = $('repoUrl').value.trim(), t = $('githubToken').value.trim();
            if (!url) { toast('لینک مخزن را وارد کنید'); return; }
            var m = url.match(/github\.com\/([^\/]+)\/([^\/]+?)(?:\.git)?$/);
            if (!m) { toast('لینک نامعتبر'); return; }
            S.username = m[1]; S.repo = m[2].replace(/\.git$/, ''); S.repoUrl = url;
            if (t) S.token = t; saveMainSettings();
            var st = $('settingsStatus'); st.innerHTML = '⚪ تست اتصال...';
            mainApi('/repos/' + S.username + '/' + S.repo).then(function() {
                st.innerHTML = '✅ متصل به ' + S.username + '/' + S.repo;
                st.style.color = '#10b981'; st.style.background = '#d1fae5'; toast('✅ اتصال موفق');
            }).catch(function(e) {
                st.innerHTML = '⚠️ ذخیره شد اما: ' + e.message;
                st.style.color = '#92400e'; st.style.background = '#fef3c7';
            });
        }
        
        function testConnection() {
            clearLog('homeLog'); showLog('homeLog'); addLog('homeLog', '🔍 تست اتصال...');
            if (!S.token || !S.username) { addLog('homeLog', '❌ تنظیمات ناقص', 'error'); return; }
            mainApi('/repos/' + S.username + '/' + S.repo).then(function() {
                addLog('homeLog', '✅ مخزن: ' + S.username + '/' + S.repo, 'success');
                return mainApi('/repos/' + S.username + '/' + S.repo + '/actions/workflows');
            }).then(function(res) { return res.json(); }).then(function(d) {
                addLog('homeLog', 'Workflowها: ' + d.total_count);
                var a = 0;
                for (var i = 0; i < d.workflows.length; i++) { if (d.workflows[i].state === 'active') a++; }
                addLog('homeLog', 'فعال: ' + a + ' | غیرفعال: ' + (d.total_count - a));
                return mainApi('/rate_limit');
            }).then(function(res) { return res.json(); }).then(function(d) {
                addLog('homeLog', 'Rate Limit: ' + d.resources.core.used + '/' + d.resources.core.limit);
            }).catch(function(e) { addLog('homeLog', '❌ ' + e.message, 'error'); });
        }
        
        function checkRuns() {
            clearLog('homeLog'); showLog('homeLog'); addLog('homeLog', '🔄 اجراهای فعال...');
            mainApi('/repos/' + S.username + '/' + S.repo + '/actions/runs?per_page=10').then(function(res) { return res.json(); })
                .then(function(d) {
                    addLog('homeLog', 'تعداد: ' + d.total_count);
                    for (var i = 0; i < d.workflow_runs.length; i++) {
                        var r = d.workflow_runs[i], icon = r.status === 'in_progress' ? '🔄' : r.status === 'queued' ? '⏳' : r.conclusion === 'success' ? '✅' : '❌';
                        addLog('homeLog', icon + ' ' + r.name + ' | ' + r.status + ' | ' + r.conclusion, r.conclusion === 'failure' ? 'error' : '');
                    }
                }).catch(function(e) { addLog('homeLog', '❌ ' + e.message, 'error'); });
        }
        
        function listWorkflows() {
            clearLog('toolsLog'); showLog('toolsLog'); addLog('toolsLog', '📡 workflowها...');
            mainApi('/repos/' + S.username + '/' + S.repo + '/actions/workflows').then(function(res) { return res.json(); })
                .then(function(d) {
                    addLog('toolsLog', 'تعداد: ' + d.total_count);
                    for (var i = 0; i < d.workflows.length; i++) {
                        var w = d.workflows[i];
                        addLog('toolsLog', (w.state === 'active' ? '✅' : '❌') + ' ' + w.name + ' [' + w.state + ']', w.state === 'active' ? 'success' : 'error');

                }).catch(function(e) { addLog('toolsLog', '❌ ' + e.message, 'error'); });

        function cancelAllRuns() {
            if (!confirm('استپ همه؟')) return;
            clearLog('toolsLog'); showLog('toolsLog'); addLog('toolsLog', '⏹️ استپ...');
            mainApi('/repos/' + S.username + '/' + S.repo + '/actions/runs?status=in_progress&status=queued').then(function(res) { return res.json(); })
                .then(function(d) {
                    if (!d.total_count) { addLog('toolsLog', '✅ هیچی فعال نیست', 'success'); return; }
                    return Promise.all(d.workflow_runs.map(function(r) {
                        return mainApi('/repos/' + S.username + '/' + S.repo + '/actions/runs/' + r.id + '/cancel', 'POST')
                            .then(function() { addLog('toolsLog', '✅ ' + r.name, 'success'); })
                            .catch(function(e) { addLog('toolsLog', '❌ ' + e.message, 'error'); });
                    }));
                }).then(function() { addLog('toolsLog', '🎯 تموم', 'success'); }).catch(function(e) { addLog('toolsLog', '❌ ' + e.message, 'error'); });
        }
